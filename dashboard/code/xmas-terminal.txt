#!/usr/bin/env node
// The operator's hand on the Xmas workshop (2026-11-17). Reads XMAS_ADMIN from .env.
//   node scripts/xmas-admin.mjs list
//   node scripts/xmas-admin.mjs order <slug> <n>           its place on the switch (smaller first; everyone uploaded is on it)
//   node scripts/xmas-admin.mjs block <slug>               gone at once: files, list, uploads
//   node scripts/xmas-admin.mjs unblock <slug>
import 'dotenv/config';
const API = process.env.BOBAI_XMAS_API || 'https://bobai-xmas.bobbuildonbnb.workers.dev';
const [action = 'list', slug, order] = process.argv.slice(2);
if (!process.env.XMAS_ADMIN) { console.error('No XMAS_ADMIN in .env — run: node scripts/xmas-admin-secret.mjs'); process.exit(1); }
const r = await fetch(API + '/admin', { method: 'POST', headers: { 'content-type': 'application/json', 'x-xmas-admin': process.env.XMAS_ADMIN },
  body: JSON.stringify({ action, slug, ...(order != null ? { order: Number(order) } : {}) }) });
const j = await r.json().catch(() => ({}));
if (!r.ok) { console.error(j.error || r.status); process.exit(1); }
if (action === 'list') for (const t of j.themes) console.log(`${t.blocked ? 'BLOCKED ' : t.ver ? 'IN      ' : 'empty   '} ${t.slug.padEnd(24)} v${t.ver}  ${String(t.files).padStart(3)} files ${(t.bytes / 1024).toFixed(0).padStart(6)} KB  "${t.title}" by ${t.builder}  ${t.preview}`);
else console.log(JSON.stringify(j.theme, null, 2));

// bobai-xmas: the community Xmas terminal's own worker (2026-11-17). Everything is in xmas.js; this worker holds no
// key, no wallet and no binding to anything else of the project — on purpose (operator: "clean and hack-free").
import { handleXmas } from './xmas.js';

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname.replace(/\/+$/, '').replace(/^\/xmas(?=\/|$)/, '') || '/';
    if (path === '/') return new Response(JSON.stringify({ name: 'BOBAI Xmas terminal workshop', how: 'npx https://brainonbnb.com/xmas/kit.tgz', themes: url.origin + '/themes' }, null, 2), { headers: { 'Content-Type': 'application/json; charset=utf-8', 'Access-Control-Allow-Origin': '*' } });
    try { return await handleXmas(request, env, url, path); }
    catch (e) { return new Response(JSON.stringify({ error: 'something broke on our side; nothing was changed' }), { status: 500, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' } }); }
  },
};

const b64bytes = (b64) => { const bin = atob(String(b64 || '')); const u = new Uint8Array(bin.length); for (let i = 0; i < bin.length; i++) u[i] = bin.charCodeAt(i); return u; };

// Checks an upload and packs it: one buffer, an index of [path, type, offset, length].
export function packUpload(files) {
  if (!Array.isArray(files) || !files.length) return { error: 'no files: send {files:[{path, b64}]}' };
  if (files.length > XMAS_LIMITS.files) return { error: `${files.length} files; at most ${XMAS_LIMITS.files}` };
  const seen = new Set(), parts = [], index = [];
  let total = 0;
  for (const f of files) {
    const c = checkPath(f?.path); if (c.error) return { error: c.error };
    if (seen.has(c.path)) return { error: `${c.path} twice` };
    seen.add(c.path);
    let bytes; try { bytes = b64bytes(f.b64); } catch { return { error: `${c.path}: b64 is not base64` }; }
    if (bytes.length > XMAS_LIMITS.fileBytes) return { error: `${c.path}: ${(bytes.length / 1048576).toFixed(1)} MB; one file at most ${XMAS_LIMITS.fileBytes / 1048576} MB` };
    index.push([c.path, c.type, total, bytes.length]); parts.push(bytes); total += bytes.length;
    if (total > XMAS_LIMITS.totalBytes) return { error: `more than ${XMAS_LIMITS.totalBytes / 1048576} MB in all` };
  }
  if (!seen.has('back.html') && !seen.has('front.html')) return { error: 'a theme needs back.html (behind BOBAI) or front.html (in front of him), or both, at the top of the folder' };
  const buf = new Uint8Array(total); let o = 0; for (const p of parts) { buf.set(p, o); o += p.length; }
  return { buf, index, total };
}

export async function handleXmas(request, env, url, path) {
  if (request.method === 'OPTIONS') return new Response(null, { status: 204, headers: CORS });
  if (!env.XMAS) return out({ error: 'storage is not bound' }, 503);

  // a file of a theme, sandboxed
  const fm = path.match(/^\/t\/([a-z0-9_-]{2,24})\/(\d{1,6})\/(.+)$/);
  if (fm && request.method === 'GET') {
    const [, slug, ver, rawPath] = fm;
    const c = checkPath(decodeURIComponent(rawPath)); if (c.error) return new Response('not found', { status: 404 });
    const t = await env.XMAS.get(themeKey(slug), 'json', { cacheTtl: 60 });
    if (!t || t.blocked) return new Response('not found', { status: 404 });
    const idx = Number(ver) === t.ver ? t.files : Number(ver) === t.prevVer ? t.prevFiles : null;
    const ent = idx?.find((f) => f[0] === c.path);
    if (!ent) return new Response('not found', { status: 404, headers: { 'Content-Security-Policy': THEME_CSP } });
    const buf = await env.XMAS.get(bundleKey(slug, ver), { type: 'arrayBuffer', cacheTtl: 3600 });
    if (!buf) return new Response('not found', { status: 404 });
    return new Response(buf.slice(ent[2], ent[2] + ent[3]), { headers: { 'Content-Type': ent[1], 'Content-Security-Policy': THEME_CSP, 'X-Content-Type-Options': 'nosniff',
      'Cache-Control': 'public, max-age=86400', 'Cross-Origin-Resource-Policy': 'cross-origin', 'Referrer-Policy': 'no-referrer', 'Access-Control-Allow-Origin': '*' } });
  }

  if (path === '/themes' && request.method === 'GET') {
    const slugs = await listSlugs(env);
    const all = (await Promise.all(slugs.map((s) => env.XMAS.get(themeKey(s), 'json', { cacheTtl: 60 })))).filter(Boolean);
    const live = all.filter((t) => !t.blocked && t.ver).sort((a, b) => (a.order ?? 999) - (b.order ?? 999) || String(a.created).localeCompare(String(b.created)));
    const use = await readUse(env);
    return out({ season: XMAS_SEASON, in_season: inSeason(), day: use.day,
      live: live.map((t) => ({ ...publicView(t), users_today: use.today[t.slug] || 0, users_total: use.total[t.slug] || 0 })),
      users_today_all: Object.values(use.today).reduce((a, b) => a + b, 0), builders: all.filter((t) => !t.blocked).length }, 200, { 'Cache-Control': 'public, max-age=60' });
  }

  // HOW MANY USE WHICH TERMINAL (operator, 2026-11-17): the terminal says once a day per browser which theme it shows.
  // A count of browsers, never who: no address, no IP kept. Counts go into one value per day (a few lost under a burst
  // of simultaneous writes is the price of not holding anything about a visitor).
  if (path === '/use' && request.method === 'POST') {
    const b = await request.json().catch(() => null), slug = String(b?.slug || '');
    if (!/^[a-z0-9_-]{2,24}$/.test(slug)) return out({ ok: false }, 400);
    const t = await env.XMAS.get(themeKey(slug), 'json', { cacheTtl: 60 });
    if (!t || t.blocked || !t.ver) return out({ ok: false, counted: false });
    const day = new Date().toISOString().slice(0, 10), k = `xmas:use:${day}`;
    const m = (await env.XMAS.get(k, 'json')) || {};
    m[slug] = (m[slug] || 0) + 1;
    await env.XMAS.put(k, JSON.stringify(m), { expirationTtl: 90 * 86400 });
    const tot = (await env.XMAS.get('xmas:use:total', 'json')) || {};
    tot[slug] = (tot[slug] || 0) + 1;
    await env.XMAS.put('xmas:use:total', JSON.stringify(tot));
    return out({ ok: true, counted: true });
  }
  const one = path.match(/^\/theme\/([a-z0-9_-]{2,24})$/);
  if (one && request.method === 'GET') {
    const t = await env.XMAS.get(themeKey(one[1]), 'json', { cacheTtl: 60 });
    if (!t || t.blocked) return out({ error: 'no such theme' }, 404);
    return out({ season: XMAS_SEASON, theme: publicView(t) }, 200, { 'Cache-Control': 'public, max-age=30' });
  }

  if (path === '/join' && request.method === 'POST') {
    const b = await request.json().catch(() => null);
    const title = cleanText(b?.title, 48), builder = cleanText(b?.builder, 32), slug = slugOf(builder);
    if (!title) return out({ error: 'title: what is your terminal called?' }, 400);
    if (!builder || !slug) return out({ error: 'builder: your name as it should show (2-24 letters or digits)' }, 400);
    const ip = request.headers.get('cf-connecting-ip') || 'x', day = new Date().toISOString().slice(0, 10), ipKey = `xmas:join:${day}:${await sha(ip)}`;
    const n = Number(await env.XMAS.get(ipKey)) || 0;
    if (n >= XMAS_LIMITS.joinsPerIpPerDay) return out({ error: 'too many new builders from here today; try tomorrow' }, 429);
    const slugs = await listSlugs(env);
    if (slugs.length >= XMAS_LIMITS.themes) return out({ error: 'the workshop is full' }, 503);
    if (slugs.includes(slug)) return out({ error: `the name "${builder}" is taken; choose another` }, 409);
    // 32 characters from 62 by rejection (no modulo bias): about 190 bits
    const A = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
    let key = 'xk_';
    while (key.length < 35) for (const x of crypto.getRandomValues(new Uint8Array(48))) if (x < 248 && key.length < 35) key += A[x % 62];
    const now = new Date().toISOString();
    await env.XMAS.put(themeKey(slug), JSON.stringify({ slug, title, builder, ver: 0, files: [], created: now, updated: null, uploads: {} }));
    await env.XMAS.put(keyKey(await sha(key)), JSON.stringify({ slug, created: now }));
    await env.XMAS.put('xmas:index', JSON.stringify([...slugs, slug]));
    await env.XMAS.put(ipKey, String(n + 1), { expirationTtl: 172800 });
    return out({ key, slug, title, builder, preview: `${SITE}/?xmas=${slug}`, note: 'Keep the key: it is you. The kit stores it in .xmas-key; never post it.' });
  }

  if (path === '/hello' && request.method === 'POST') {
    const r = await builderOf(request, env); if (r.error) return r.error;
    return out({ theme: publicView(r.t), season: XMAS_SEASON, limits: XMAS_LIMITS });
  }

  if (path === '/upload' && request.method === 'POST') {
    const r = await builderOf(request, env); if (r.error) return r.error;
    const t = r.t, day = new Date().toISOString().slice(0, 10);
    const used = Number(t.uploads?.[day]) || 0;
    if (used >= XMAS_LIMITS.uploadsPerDay) return out({ error: `${XMAS_LIMITS.uploadsPerDay} uploads today already; the dev page shows your changes without uploading` }, 429);
    const b = await request.json().catch(() => null);
    const p = packUpload(b?.files); if (p.error) return out({ error: p.error }, 400);
    const ver = (t.ver || 0) + 1;
    await env.XMAS.put(bundleKey(t.slug, ver), p.buf.buffer);
    const old = t.prevVer;
    const title = cleanText(b?.title, 48) || t.title;
    const next = { ...t, title, prevVer: t.ver || null, prevFiles: t.files || [], ver, files: p.index, bytes: p.total, updated: new Date().toISOString(), uploads: { [day]: used + 1 } };
    await env.XMAS.put(themeKey(t.slug), JSON.stringify(next));
    if (old) await env.XMAS.delete(bundleKey(t.slug, old)).catch(() => {});   // the version before the previous one
    return out({ ok: true, ver, files: p.index.length, bytes: p.total, preview: `${SITE}/?xmas=${t.slug}`, live: true,
      note: 'Uploaded. The preview link shows it to anyone; from 1 December it is on the terminal\'s switch under your name (every upload replaces the last).' });
  }

  if (path === '/admin' && request.method === 'POST') {
    if (!env.XMAS_ADMIN || request.headers.get('x-xmas-admin') !== env.XMAS_ADMIN) return out({ error: 'unauthorized' }, 401);
    const b = await request.json().catch(() => ({}));
    const slugs = await listSlugs(env);
    if (b.action === 'list') {
      const all = (await Promise.all(slugs.map((s) => env.XMAS.get(themeKey(s), 'json')))).filter(Boolean);
      return out({ themes: all.map((t) => ({ ...publicView(t), blocked: !!t.blocked, order: t.order ?? null, bytes: t.bytes || 0, files: (t.files || []).length, created: t.created })) });
    }
    const t = await env.XMAS.get(themeKey(String(b.slug || '')), 'json');
    if (!t) return out({ error: 'no such theme' }, 404);
    if (b.action === 'block') { t.blocked = true; t.blockedAt = new Date().toISOString(); }
    else if (b.action === 'unblock') t.blocked = false;
    else if (b.action === 'order') t.order = Number(b.order);
    else if (b.action === 'rename') { t.title = cleanText(b.title, 48) || t.title; }
    else return out({ error: 'action: list | block | unblock | order | rename' }, 400);
    await env.XMAS.put(themeKey(t.slug), JSON.stringify(t));
    return out({ ok: true, theme: { ...publicView(t), blocked: !!t.blocked, order: t.order ?? null } });
  }
  return out({ error: 'not found', see: 'POST /join, /hello, /upload, /use; GET /themes, /theme/<slug>' }, 404);
}

You are helping **{{BUILDER}}** build **"{{TITLE}}"**: their own Xmas look for the **$BOBAI Brain Terminal** on
https://brainonbnb.com — the live terminal where BOBAI, the project's mascot, shows every trade, burn and liquidity
add of the $BOBAI token on BNB Chain as it happens. This folder is everything you need. Read this file once, fully;

From **1 December 2026** visitors pick an Xmas terminal by its builder's name at the snowflake next to LIVE, and the
terminal counts how many use each one, every day. Anyone may build one: people, teams, platforms.

## What you build — and what stays

- **You build the surroundings and your own animations:** a scene, a sky, snow, lights, decorations around the
  stations, things that react when somebody buys or a burn happens. Your style, your idea. Nothing is given to you
  and nothing is expected: you start from two empty pages.
- **What always stays as it is:** BOBAI himself (his figure, moves and words), the terminal's labels, numbers, log,
  timeline, buttons and windows. Your pages lie in layers *between* them; they cannot cover a button or take a click.
- **The standard terminal is always one click away** for every visitor.

## The two pages

| file | where it lies | use it for |
|---|---|---|
| `theme/back.html` | behind BOBAI, over the terminal's 3D scene | the world: sky, ground, scenery, things around the stations |
| `theme/front.html` | in front of BOBAI, under every label, button and window | an overlay: snow, sparkles, things flying by |

Either one or both. Both fill the terminal's window (`100% × 100%`) and are **transparent** unless you paint. In
`front.html` never paint a full opaque background: it is an overlay, not a wall. In `back.html` you may paint a whole
scene; the 3D stations of the terminal are then behind it — the `layout` message tells you where each one is, so you
can decorate them in your own way instead.

Put every file you load (images, scripts, fonts, video, 3D models) inside `theme/` and load it with a relative path:
`<img src="img/tree.webp">`, `<script src="snow.js"></script>`, `import * as THREE from './lib/three.module.min.js'`.

## What the terminal tells your pages

Every message arrives as `window.addEventListener('message', e => …)` with `e.data`:
`{ bobai: 'xmas', v: 1, type: …, … }`. Ignore anything without `bobai: 'xmas'`. Positions are CSS pixels inside the
terminal's window, `x`/`y` from its top-left; a box is `{ x, y, w, h }`.

## The sandbox (what does not work, on purpose)

Your pages run in a sandbox, in dev exactly as live:

- **no network**: `fetch`, XHR, WebSocket, beacons and files from other sites are blocked → everything in `theme/`
- **no `eval`, `new Function`, WebAssembly, web workers** → plain scripts and modules
- **no cookies, `localStorage`, `sessionStorage`, IndexedDB** (they throw) → keep state in variables
- **no sound**, no forms, no popups, no frames, no clicks (the layers take no pointer events)
- **no talking back**: the terminal tells your pages things and listens to nothing they say

`node xmas.mjs check` warns about all of this before you upload.

## Fast, calm, on every screen

- The terminal measures its own frame rate while a theme is on. **Under 20 fps for 8 seconds, the theme is put away
  for that visitor** and the standard terminal stays. Phones first: test the dev page at 390 px wide (browser dev
  tools, device mode) and on a real phone if you can.
- Animate with `requestAnimationFrame`, CSS `transform`/`opacity`, or one `<canvas>`; cap the canvas at
  `Math.min(devicePixelRatio, 2)`; prefer WebP images and short muted looping video; a few hundred particles, not
  tens of thousands.
- **Stop when `visible` says `on: false`** (and on `document.hidden`).
- **`reduced: true`** (the visitor asked their device for less motion): keep it still or nearly still — no flashing,
  no fast movement, no shaking. Never flash the whole screen for anyone (photosensitivity).
- Leave BOBAI's face and the terminal's boxes (`layout.ui`, `layout.figure`) clear of busy motion in `front.html`.

## Pictures and clips — made the way the BOBAI terminal makes its own

The terminal's own pictures and clips are made with OpenAI's image models and Google's Veo. You can do the same, with
**your human's own keys and credits**: they go into `.env` in this folder (`OPENAI_API_KEY=…`, `GEMINI_API_KEY=…`),
stay on this machine and are sent only to OpenAI or Google. Never put a key into `theme/` — `check` and `upload`
refuse any file that contains one.

```
node xmas.mjs image "a cozy gingerbread house, warm window light, 3D animated-film look" --out theme/img/house.webp
     --size 1024x1024 | 1536x1024 | 1024x1536     --quality low | medium | high     --opaque (default: transparent)
     --ref theme/img/house.webp   (a new picture in the same style / a frame of a small animation: "the same house, now with snow on the roof")
     --model …  (default: the newest image model the key can use that draws what you asked; transparent needs one that supports it)
node xmas.mjs video "snow falls slowly over a quiet pine forest at night, the camera does not move" --out theme/vid/forest.mp4
     --ratio 16:9 | 9:16    --secs 8    --image theme/img/start.png (first frame)    --last theme/img/end.png (last frame; same as --image = a seamless loop)
     --model veo-3.1-fast-generate-preview (default) | veo-3.1-generate-preview
```

What works well (learned on the terminal): one style for everything, written into every prompt; transparent PNG/WebP
for things that sit on the scene; a reference picture (`--ref`) to keep a set of pictures matching; for a seamless
looping clip give the same picture as first and last frame. A clip that must be cut out (a figure over the scene) is
drawn on a flat green or blue screen and keyed in a `<canvas>` or a small WebGL shader — **set `video.crossOrigin = 'anonymous'`**
(and `img.crossOrigin` for pictures you read pixels from): your pages are their own origin, and without it the canvas
cannot read the frames; a whole backdrop in
`back.html` needs no keying. Videos play **muted** (`<video autoplay muted loop playsinline>`): the terminal has no
sound, and a video with sound will not start. Keep clips short and small (720p, a few MB); turn large PNGs into WebP.
Ask your human before spending credits on many pictures or clips.

webm, woff/woff2, glb/gltf/bin.

## How it goes live

There is no review and no waiting: **every uploaded theme is on the terminal's switch from 1 December**, under its
builder's name, with how many visitors use it today. Every upload replaces the last one.

**Everyone may build one** — a builder, a team, another memecoin, a platform, an exchange. Your own name, logo or
mascot in your theme is fine: it is your theme. The rules:

- **no hate, no racism, no terror** — nothing that attacks people for who they are, nothing that glorifies violence
  or terrorism. This is the hard line: a theme that crosses it is stopped at once.
- **NSFW within limits is allowed**: crude jokes, memes, suggestive is fine; no explicit sex or nudity, nothing
  involving minors, nobody's private data.
- **no scam**: no wallet addresses, no "send", no QR codes, no payment requests, no price promises ("100x"), no fake
  giveaways — the terminal shows the chain; a theme that looks like a payment request is refused.
- it does not pretend to be the terminal's own buttons, numbers or warnings, or to be somebody it is not.
- it works on a phone and a desktop and stays fast (see above).

A theme can be stopped at any time, also after 1 December. Your key (`.xmas-key`) is you: whoever has it can change
your theme. Never commit it, post it or paste it anywhere.

## When your human is stuck (you fix it, they should not have to)

Assume your human may have never used a terminal. Do the typing yourself, explain in one plain sentence what you do,
and never paste their key anywhere.

| what happens | what you do |
|---|---|
| they downloaded kit.tgz in the browser and "nothing opens" | the file does nothing by itself: the command must run in a terminal (you can run it for them) |
| `npx` / `node` is "not recognized" | Node.js is missing: they install the LTS from https://nodejs.org (one installer, default options), open a new terminal, run the command again |
| the start asks two questions and you cannot answer them (no keyboard in your shell) | ask your human both, then run it with `--title "…" --builder "…"` |
| "the name … is taken" | ask for another builder name (it shows on the terminal) |
| "too many new builders from here today" | wait until tomorrow (UTC), or continue in the folder that already exists |
| `node xmas.mjs dev`: "Port 5199 is busy" | `node xmas.mjs dev --port 5200` |
| the dev page opened but shows the terminal without their theme | look at the snowflake next to LIVE: it must say DEV. Open exactly the address the dev command printed (http://127.0.0.1:…/?xmas-dev=1), not brainonbnb.com |
| their theme is invisible | the page is probably transparent and empty, or drawn behind something: give `body` a test colour for a moment; check the browser console (F12) of the dev page for errors |
| a picture or script does not load | it must be inside `theme/` with a relative path; nothing from other sites loads (`node xmas.mjs check` names it) |
| it worked, then stopped moving | the theme went over budget (under 20 fps for 8 s) and the terminal put it away: fewer particles, smaller canvas, no huge images |
| `image` / `video`: "No OPENAI_API_KEY / GEMINI_API_KEY" | they paste their own key after the `=` in `.env` in the folder; OpenAI keys: platform.openai.com → API keys; Google: aistudio.google.com → Get API key |
| `image`: an error about the model or "verification" | the kit already tries every image model the key can use; some OpenAI accounts must verify the organisation (platform.openai.com → settings) for the newest ones |
| `video`: "quota" / "billing" / 429 | the Google key needs billing (or credit) for Veo; one clip costs roughly a dollar or more |
| `upload`: "contains what looks like an API key" | a key ended up inside `theme/`: remove it, keys live only in `.env` |
| `upload`: 401 "unknown key" / "no key" | `.xmas-key` is missing or changed: it is in the folder the start command made; without it the theme cannot be changed (start a new one with another name) |
| `upload`: 429 | 120 uploads a day: use the dev page for trying things, upload when it is good |
| Windows: quotes in prompts break | use double quotes around the prompt; no line breaks inside it |
| they want to start over | delete the files in `theme/` and build again; the name and key stay theirs |

Anything else: read the error out loud to them in plain words, and look at `node xmas.mjs check`. The kit is a single
file, `xmas.mjs`, in this folder — you may read it to see exactly what a command does.

## Working with your human

1. If you have not yet: run `node xmas.mjs dev` and let them see the empty terminal.
2. Ask what they imagine: the mood, the colours, what should happen when someone buys or a burn hits. Offer two or

3. Build in small steps; after each one, they look at the dev page.
4. Before you call it done: `node xmas.mjs check`, a phone-width look, the console free of errors, and one test of each
   event (above). Then `node xmas.mjs upload` and give them the preview link.

// ---------------------------------------------------------------- the theme folder
function walk(root) {
  const out = [];
  const go = (d) => { for (const e of fs.readdirSync(d, { withFileTypes: true })) { if (e.name.startsWith('.')) continue; const p = path.join(d, e.name); if (e.isDirectory()) go(p); else if (e.isFile()) out.push(p); } };
  if (fs.existsSync(root)) go(root);
  return out;
}
function checkTheme(root) {
  const files = walk(root), problems = [], warnings = [];
  let total = 0;
  const rels = files.map((f) => path.relative(root, f).split(path.sep).join('/'));
  if (!rels.includes('back.html') && !rels.includes('front.html')) problems.push('theme/ needs back.html (behind BOBAI) or front.html (in front of him)');
  if (files.length > LIMITS.files) problems.push(`${files.length} files; at most ${LIMITS.files}`);
  files.forEach((f, i) => {
    const rel = rels[i], ext = (rel.match(/\.([A-Za-z0-9]+)$/) || [])[1]?.toLowerCase(), size = fs.statSync(f).size;
    total += size;
    if (!/^[A-Za-z0-9][A-Za-z0-9._/-]{0,119}$/.test(rel)) problems.push(`${rel}: use letters, digits, . _ - and / in file names`);
    if (!ext || !TYPES[ext]) problems.push(`${rel}: .${ext || '?'} is not served (html, css, js, json, images, video, fonts, glb)`);
    if (size > LIMITS.fileBytes) problems.push(`${rel}: ${(size / 1048576).toFixed(1)} MB; one file at most 5 MB`);
    if (/\.(html?|css|m?js)$/i.test(rel)) {
      const t = fs.readFileSync(f, 'utf8');
      const ext2 = t.match(/(?:src|href)\s*=\s*["']\s*(https?:)?\/\/[^"']+|url\(\s*["']?(https?:)?\/\//gi);
      if (ext2) warnings.push(`${rel}: loads from another site (${ext2[0].slice(0, 60)}…) — the terminal blocks it; put the file in theme/`);
      if (/\bfetch\s*\(|XMLHttpRequest|WebSocket|EventSource|sendBeacon/.test(t)) warnings.push(`${rel}: calls out (fetch/XHR/WebSocket) — blocked on the terminal; everything must be in theme/`);
      if (/\beval\s*\(|new\s+Function\s*\(|WebAssembly/.test(t)) warnings.push(`${rel}: eval / new Function / WebAssembly — blocked on the terminal`);
      if (/<audio|new\s+Audio\s*\(|AudioContext/.test(t)) warnings.push(`${rel}: sound — the terminal has none; it will not play`);
      if (/localStorage|sessionStorage|indexedDB|document\.cookie/.test(t)) warnings.push(`${rel}: storage/cookies — not available in the sandbox (it throws); keep state in variables`);
    }
  });
  if (total > LIMITS.totalBytes) problems.push(`${(total / 1048576).toFixed(1)} MB in all; at most 20 MB`);
  // NEVER A KEY IN AN UPLOAD: an API key (OpenAI, Google, any sk-/AIza- form) or the kit's own key in a file of theme/
  // stops the upload — whatever is uploaded can be read by anyone with the preview link
  const own = (() => { try { return fs.readFileSync(path.join(path.dirname(root), '.xmas-key'), 'utf8').trim(); } catch { return null; } })();
  files.forEach((f, i) => {
    if (!/\.(html?|css|m?js|json|txt|svg|gltf)$/i.test(rels[i])) return;
    const t = fs.readFileSync(f, 'utf8');
    if (/\bsk-(proj-|svcacct-|admin-)?[A-Za-z0-9_-]{20,}|\bAIza[0-9A-Za-z_-]{30,}|\bxk_[A-Za-z0-9]{32}\b|OPENAI_API_KEY\s*=\s*\S|GEMINI_API_KEY\s*=\s*\S/.test(t) || (own && t.includes(own)))
      problems.push(`${rels[i]}: contains what looks like an API key or your .xmas-key — never upload a key; keys stay in .env`);
  });
  return { files, rels, total, problems, warnings };
}

}

// ---------------------------------------------------------------- dev: the sandbox page, mirrored, with your folder in it
async function dev(dir) {
  const { cfg, theme } = readProject(dir);
  const port = Number(flag('port')) || 5199;
  const EMPTY = '<!doctype html><meta charset="utf-8"><title></title>';
  let ver = 0, sig = '';
  const scan = () => { try { const s = walk(theme).map((f) => { const st = fs.statSync(f); return f + st.size + ':' + st.mtimeMs; }).join('|'); if (s !== sig) { sig = s; ver++; } } catch {} };
  scan(); setInterval(scan, 600);
  const server = http.createServer(async (req, res) => {
    try {
      const u = new URL(req.url, 'http://127.0.0.1');
      if (u.pathname.startsWith('/__theme/')) {
        if (u.pathname === '/__theme/__version') { res.writeHead(200, { 'content-type': 'text/plain', 'cache-control': 'no-store' }); return res.end(String(ver)); }
        const rel = decodeURIComponent(u.pathname.slice('/__theme/'.length));
        const file = path.resolve(theme, rel);
        if (!file.startsWith(path.resolve(theme) + path.sep)) { res.writeHead(404); return res.end(); }
        const ext = (rel.match(/\.([A-Za-z0-9]+)$/) || [])[1]?.toLowerCase();
        const headers = { 'content-security-policy': THEME_CSP, 'x-content-type-options': 'nosniff', 'cache-control': 'no-store', 'access-control-allow-origin': '*' }; // as live: a canvas may read a video or picture loaded with crossOrigin
        if (!fs.existsSync(file)) { if (/^(back|front)\.html$/.test(rel)) { res.writeHead(200, { ...headers, 'content-type': TYPES.html }); return res.end(EMPTY); } res.writeHead(404, headers); return res.end(); }
        if (!ext || !TYPES[ext]) { res.writeHead(415, headers); return res.end('not a kind of file the terminal serves'); }
        res.writeHead(200, { ...headers, 'content-type': TYPES[ext] }); return res.end(fs.readFileSync(file));
      }
      // everything else: the sandbox page itself, fetched for you
      const body = ['GET', 'HEAD'].includes(req.method) ? undefined : await new Promise((r) => { const c = []; req.on('data', (d) => c.push(d)); req.on('end', () => r(Buffer.concat(c))); });
      const up = await fetch(SANDBOX + u.pathname + u.search, { method: req.method, redirect: 'manual', body, headers: { accept: req.headers.accept || '*/*', 'content-type': req.headers['content-type'] || '', 'user-agent': 'bobai-xmas-kit-dev/1' } });
      const h = {};
      up.headers.forEach((v, k) => { if (!['content-encoding', 'content-length', 'transfer-encoding', 'connection', 'strict-transport-security', 'alt-svc'].includes(k)) h[k] = v; });
      if (h['content-security-policy']) h['content-security-policy'] = h['content-security-policy'].replace(/;\s*upgrade-insecure-requests/g, '');
      if (h.location) h.location = h.location.replace(SANDBOX, `http://127.0.0.1:${port}`);
      res.writeHead(up.status, h); res.end(Buffer.from(await up.arrayBuffer()));
    } catch (e) { res.writeHead(502, { 'content-type': 'text/plain' }); res.end('the sandbox page could not be reached: ' + e.message); }
  });

// Gemini Omni: one request to /interactions (a first and a last image, the prompt, "a video"), asked again until the
// clip is there, then the file downloaded — the same way the BOBAI terminal's own clips are made since 25 Oct 2026
async function omniVideo(key, model, out, prompt, dir) {

    { type: 'text', text: prompt + ` The clip is exactly ${secs} seconds long${flag('image') && flag('last') ? '; it starts on the first image and ends exactly on the second image' : ''}. No audio needed.` }];

  const r = await fetch(`${API}/interactions`, { method: 'POST', headers: H, body: JSON.stringify({ model, input, response_format: { type: 'video', aspect_ratio: ratio, resolution: '720p', delivery: 'uri' } }) });
  let j = await r.json().catch(() => ({})); if (!r.ok) die(j.error?.message || 'Google answered ' + r.status);
  const find = (o, pred) => { if (o && typeof o === 'object') for (const [k, v] of Object.entries(o)) { if (pred(k, v)) return v; const f = find(v, pred); if (f) return f; } return null; };
  const uriOf = (o) => find(o, (k, v) => k === 'uri' && typeof v === 'string' && v.includes('/files/')), dataOf = (o) => find(o, (k, v) => k === 'data' && typeof v === 'string' && v.length > 100000);
  const id = j.id || j.name;
  for (let i = 0; i < 120 && !uriOf(j) && !dataOf(j); i++) {
    if (/fail|cancel|error/i.test(String(j.status || j.state || ''))) die('the clip failed: ' + JSON.stringify(j).slice(0, 300));
    await new Promise((ok) => setTimeout(ok, 10000)); process.stdout.write('.');
    try { j = await (await fetch(`${API}/interactions/${id}`, { headers: H })).json(); } catch {}
  }
  const uri = uriOf(j);
  if (uri) {
    const fid = uri.split('/files/')[1].split(/[:?]/)[0];
    for (let i = 0; i < 120; i++) { let f = {}; try { f = await (await fetch(`${API}/files/${fid}`, { headers: H })).json(); } catch {} if (f.state === 'ACTIVE') break; if (f.state === 'FAILED') die('the clip file failed'); await new Promise((ok) => setTimeout(ok, 10000)); }
    const v = await fetch(`${API}/files/${fid}:download?alt=media`, { headers: H, redirect: 'follow' });
    fs.writeFileSync(out, Buffer.from(await v.arrayBuffer()));
  } else if (dataOf(j)) fs.writeFileSync(out, Buffer.from(dataOf(j), 'base64'));
  else die('no clip came back (filtered?): ' + JSON.stringify(j).slice(0, 300));
  say(`\n  ✅  ${path.relative(dir, out)} (${(fs.statSync(out).size / 1024).toFixed(0)} KB) — the terminal plays every video muted`);
}
